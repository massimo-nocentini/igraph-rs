6/// Raw FFI bindings to the igraph C library, generated by `bindgen` at build
7/// time from the installed headers.
8///
9/// Everything here is re-exported at the crate root as well, so that
10/// `igraph::igraph_t` and `igraph::ffi::igraph_t` are the same type. The
11/// Rusty API is built *on top of these very types*: e.g. [`Graph`] is an
12/// alias of [`igraph_t`] enriched with safe methods, and owned
13/// types such as [`VectorInt`](crate::vector::VectorInt) free their memory on drop.
14#[allow(
15    non_upper_case_globals,
16    non_camel_case_types,
17    non_snake_case,
18    missing_docs,
19    dead_code,
20    unsafe_op_in_unsafe_fn,
21    clippy::all
22)]
23pub mod ffi {
24    include!(concat!(env!("OUT_DIR"), "/bindings.rs"));
25}

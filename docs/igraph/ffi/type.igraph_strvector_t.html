pub fn set(&mut self, index: usize, value: &str) -> Result<()>

Replaces the string at index with a copy of value (igraph_strvector_set_len).

use igraph::prelude::*;
let mut names = StrVector::with_len(3);
names.set(1, "hub").unwrap();
assert_eq!(names.to_vec(), vec!["", "hub", ""]);
assert_eq!(&names[1], "hub");
assert!(names.set(3, "nope").is_err());
